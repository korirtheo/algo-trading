# Post-close daily reconcile cron for the algotrader container.
#
# Installed on the AWS host by deploy/install_cron.sh. Runs the reconcile
# script INSIDE the algotrader container so it uses the same image/deps/DB.
#
# Schedule: 20:15 ET (8:15pm) Mon-Fri — after extended hours close, when
# Alpaca SIP allows querying the completed trading day. 15m buffer past 8pm.
#
# Ubuntu's Vixie cron ignores TZ=/CRON_TZ in /etc/cron.d. Run a cheap check
# each UTC minute and launch only at 20:15 America/New_York, including DST.
# Cron treats % specially, hence the escaped format directives below.
# Format: <minute> <hour> * * <dow> root command
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

* * * * * root stamp="$(TZ=America/New_York date +\%u-\%H:\%M)"; case "$stamp" in [1-5]-20:15) docker exec algotrader python /app/scripts/reconcile/post_close_reconcile.py ;; esac >> /home/ubuntu/algo-trading/logs/reconcile_cron.log 2>&1
